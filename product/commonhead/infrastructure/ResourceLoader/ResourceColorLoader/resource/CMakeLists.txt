include(TemplateTools)
include(CapitalizeString)
# 设置资源字符串 JSON 文件路径

set(RESOURCE_IDE_FOLDER commonhead/codegen)
set(MODULE_HEADER_BASE_DIR ${UCF_CODE_GENERATOR_DIR})
set(MODULE_BUILD_INTERFACE_DIR ${MODULE_HEADER_BASE_DIR})
set(MODULE_INSTALL_INTERFACE_DIR include)
set(RESOURCE_OUTPUT_DIR
    ${MODULE_HEADER_BASE_DIR}/commonhead/ResourceColorLoader)
set(COLOR_TOKEN_JSON_PATH ${CMAKE_CURRENT_SOURCE_DIR}/data/token.json)

set(RESOURCE_TARGETS)
set(RESOURCE_HEADER_FILES)

set(THEME_JSON_NAMES light dark)
set(THEME_JSON_PATH_SET)
foreach(theme_name IN LISTS THEME_JSON_NAMES)
    list(APPEND THEME_JSON_PATH_SET
        "${CMAKE_CURRENT_SOURCE_DIR}/data/themes/${theme_name}.json")
endforeach()

## Precheck: token ↔ theme 一致性校验
include(ValidateColorTokens)
validate_color_tokens(
    SCRIPT     ${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate_color_tokens.py
    TOKEN_FILE ${COLOR_TOKEN_JSON_PATH}
    THEME_FILES ${THEME_JSON_PATH_SET}
    PALETTE_FILE ${CMAKE_CURRENT_SOURCE_DIR}/data/palette.json
    OUTPUT_TARGET_VAR PRECHECK_TARGET
)
list(APPEND RESOURCE_TARGETS "${PRECHECK_TARGET}")

## ColorPalette
set(COLOR_JSON_PATH ${CMAKE_CURRENT_SOURCE_DIR}/data/palette.json)
set(COLOR_OUTPUT_FILE ColorPaletteModel.h)
set(COLOR_OUTPUT_PATH ${RESOURCE_OUTPUT_DIR}/${COLOR_OUTPUT_FILE})
set(COLOR_TEMPLATE_FILE ${CMAKE_CURRENT_SOURCE_DIR}/templates/ColorPaletteModel.h.j2)

generate_from_template(
    TEMPLATE_FILE ${COLOR_TEMPLATE_FILE}
    INPUT_FILE ${COLOR_JSON_PATH}
    OUTPUT_FILE ${COLOR_OUTPUT_PATH}
    OUTPUT_TARGET_VAR COLOR_TARGET
    DEPENDS ${PRECHECK_TARGET}
)
add_dependencies("${COLOR_TARGET}" "${PRECHECK_TARGET}")
list(APPEND RESOURCE_TARGETS "${COLOR_TARGET}")
list(APPEND RESOURCE_HEADER_FILES "${COLOR_OUTPUT_PATH}")

## ColorToken
set(TOKEN_OUTPUT_FILE ColorToken.h)
set(TOKEN_OUTPUT_PATH ${RESOURCE_OUTPUT_DIR}/${TOKEN_OUTPUT_FILE})
set(TOKEN_TEMPLATE_FILE ${CMAKE_CURRENT_SOURCE_DIR}/templates/ColorToken.h.j2)

generate_from_template(
    TEMPLATE_FILE ${TOKEN_TEMPLATE_FILE}
    INPUT_FILE ${COLOR_TOKEN_JSON_PATH}
    OUTPUT_FILE ${TOKEN_OUTPUT_PATH}
    OUTPUT_TARGET_VAR TOKEN_TARGET
    DEPENDS ${PRECHECK_TARGET}
)
add_dependencies("${TOKEN_TARGET}" "${PRECHECK_TARGET}")
list(APPEND RESOURCE_TARGETS "${TOKEN_TARGET}")
list(APPEND RESOURCE_HEADER_FILES "${TOKEN_OUTPUT_PATH}")


## BuildThemeColorSet
foreach(theme_name IN LISTS THEME_JSON_NAMES)
    set(THEME_JSON_PATH
        ${CMAKE_CURRENT_SOURCE_DIR}/data/themes/${theme_name}.json)
    capitalize(${theme_name} capital_json_name)
    set(THEME_OUTPUT_FILE Build${capital_json_name}ThemeColorSet.h)
    set(THEME_OUTPUT_PATH ${RESOURCE_OUTPUT_DIR}/${THEME_OUTPUT_FILE})
    set(THEME_TEMPLATE_FILE
        ${CMAKE_CURRENT_SOURCE_DIR}/templates/BuildThemeColorSet.h.j2)

    generate_from_template(
        TEMPLATE_FILE ${THEME_TEMPLATE_FILE}
        INPUT_FILE ${THEME_JSON_PATH}
        OUTPUT_FILE ${THEME_OUTPUT_PATH}
        OUTPUT_TARGET_VAR THEME_TARGET
        DEPENDS ${PRECHECK_TARGET}
    )
    add_dependencies("${THEME_TARGET}" "${PRECHECK_TARGET}")
    list(APPEND RESOURCE_TARGETS "${THEME_TARGET}")
    list(APPEND RESOURCE_HEADER_FILES "${THEME_OUTPUT_PATH}")
endforeach()


include(BuildInterface)
BuildInterface(
    MODULE_NAME ResourceColor
    TARGET_SOURCE_HEADER_BASE_DIR ${MODULE_HEADER_BASE_DIR}
    TARGET_SOURCE_PUBLIC_HEADER ${RESOURCE_HEADER_FILES}
    TARGET_ADD_LINK_LIBRARY_PUBLIC ResourceColorLoaderTypes
    TARGET_ADD_DEPENDENCIES ${RESOURCE_TARGETS}
    TARGET_INCLUDE_DIRECTORIES_BUILD_INTERFACE ${MODULE_BUILD_INTERFACE_DIR}
    TARGET_INCLUDE_DIRECTORIES_INSTALL_INTERFACE ${MODULE_INSTALL_INTERFACE_DIR}
    IDE_FOLDER ${RESOURCE_IDE_FOLDER}
)

set_property(
    TARGET ResourceColor
    PROPERTY UCF_RESOURCE_TOKEN_FILE "${COLOR_TOKEN_JSON_PATH}"
)
