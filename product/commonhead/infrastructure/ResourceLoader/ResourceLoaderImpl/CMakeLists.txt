include(BuildModule)

set(MODULE_MODULE_NAME ResourceLoaderImpl)
set(MODULE_IDE_FOLDER commonhead/infrastructure/ResourceLoader)
set(MODULE_HEADER_BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/include)

set(MODULE_HEADER_FILES
    include/commonhead/ResourceLoader/ResourceLoaderImplExport.h
    include/commonhead/ResourceLoader/ResourceLoaderCreator.h
)
set(MODULE_SOURCE_PRIVATE
    src/LoggerDefine.h
    src/ResourceLoader.h
    src/ResourceLoader.cpp
)

set(MODULE_PUBLIC_DEFINITIONS)
set(MODULE_PRIVATE_DEFINITIONS)
if (COMMONHEAD_LIB_TYPE STREQUAL "STATIC_LIB")
    set(MODULE_PUBLIC_DEFINITIONS RESOURCE_LOADER_IMPL_STATIC)
else()
    set(MODULE_PRIVATE_DEFINITIONS RESOURCE_LOADER_IMPL_SHARED)
endif()

BuildModule(
    ${COMMONHEAD_LIB_TYPE}
    MODULE_NAME ${MODULE_MODULE_NAME}
    TARGET_SOURCE_PRIVATE ${MODULE_SOURCE_PRIVATE}
    TARGET_SOURCE_HEADER_BASE_DIR ${MODULE_HEADER_BASE_DIR}
    TARGET_SOURCE_PUBLIC_HEADER ${MODULE_HEADER_FILES}
    TARGET_ADD_LINK_LIBRARY_PUBLIC ResourceLoaderApi
    TARGET_ADD_LINK_LIBRARY_PRIVATE
        ResourceColorLoaderImpl
        ResourceFontLoaderImpl
        ResourceAssetLoaderImpl
        ClientInfoServiceApi
        CoreFrameworkApi
        SystemUtils
        LoggerUtils
    TARGET_INCLUDE_DIRECTORIES_BUILD_INTERFACE ${MODULE_HEADER_BASE_DIR}
    TARGET_INCLUDE_DIRECTORIES_INSTALL_INTERFACE include
    TARGET_INCLUDE_DIRECTORIES_PRIVATE src
    TARGET_PUBLIC_DEFINITIONS ${MODULE_PUBLIC_DEFINITIONS}
    TARGET_PRIVATE_DEFINITIONS ${MODULE_PRIVATE_DEFINITIONS}
    IDE_FOLDER ${MODULE_IDE_FOLDER}
)
