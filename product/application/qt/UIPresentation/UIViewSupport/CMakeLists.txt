include(BuildQtModule)

if(WIN32)
    set(ADD_DEFINITIONS UIViewSupport_DLL)
else()
    set(ADD_DEFINITIONS)
endif()

set(MODULE_MODULE_NAME UIViewSupport)
set(MODULE_IDE_FOLDER UI/UIPresentation/UIViewSupport)

set(MODULE_HEADER_BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/include)

set(MODULE_HEADER_FILES
    include/UIViewSupport/UIViewSupportExport.h
    include/UIViewMessageBox/UIViewMessageBoxHelper.h
)

set(MODULE_SOURCE_PRIVATE
    src/LoggerDefine.h
    src/UIViewMessageBoxHelper.cpp
)

set(MODULE_DEPENDENCIES_PUBLIC
    Qt6::Core
    UTComposite
)

set(MODULE_DEPENDENCIES_PRIVATE
    AppContext
    LoggerUtils
    UIFabrication
    UIQmlUtilities
    UIWindowUtilities
    Qt6::Gui
    Qt6::Quick
)

BuildQtModule(
    MODULE_NAME ${MODULE_MODULE_NAME}
    TARGET_SOURCE_PRIVATE ${MODULE_SOURCE_PRIVATE}
    TARGET_SOURCE_HEADER_BASE_DIR ${MODULE_HEADER_BASE_DIR}
    TARGET_SOURCE_PUBLIC_HEADER ${MODULE_HEADER_FILES}
    TARGET_ADD_LINK_LIBRARY_PRIVATE ${MODULE_DEPENDENCIES_PRIVATE}
    TARGET_ADD_LINK_LIBRARY_PUBLIC ${MODULE_DEPENDENCIES_PUBLIC}
    TARGET_INCLUDE_DIRECTORIES_BUILD_INTERFACE ${MODULE_HEADER_BASE_DIR}
    TARGET_INCLUDE_DIRECTORIES_INSTALL_INTERFACE include
    TARGET_INCLUDE_DIRECTORIES_PRIVATE src
    TARGET_PRIVATE_DEFINITIONS ${ADD_DEFINITIONS}
    IDE_FOLDER ${MODULE_IDE_FOLDER}
)
