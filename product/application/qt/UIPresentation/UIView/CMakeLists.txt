include(BuildQtModule)
include(cmake/UIViewBuildInputs.cmake)

set(UIVIEW_BUILD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")

set_property(DIRECTORY "${UIVIEW_BUILD_ROOT}" PROPERTY UIVIEW_BUILD_SOURCES "")
set_property(DIRECTORY "${UIVIEW_BUILD_ROOT}" PROPERTY UIVIEW_BUILD_QML_FILES "")
set_property(DIRECTORY "${UIVIEW_BUILD_ROOT}" PROPERTY UIVIEW_BUILD_QML_ALIAS_MAPPINGS "")
set_property(DIRECTORY "${UIVIEW_BUILD_ROOT}" PROPERTY UIVIEW_BUILD_PRIVATE_INCLUDE_DIRS "")
set_property(DIRECTORY "${UIVIEW_BUILD_ROOT}" PROPERTY UIVIEW_BUILD_PRIVATE_DEPENDENCIES "")

if (WIN32)
    set(ADD_DEFINITIONS UIView_DLL)
else()
    set(ADD_DEFINITIONS)
endif()

set(MODULE_MODULE_NAME UIView)
set(MODULE_IDE_FOLDER UI/UIPresentation/UIView)

set(MODULE_HEADER_BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/include)

set(MODULE_HEADER_FILES
    include/UIView/UIViewExport.h
    include/UIView/UIViewModule.h
)

set(MODULE_SOURCE_PRIVATE
    src/LoggerDefine.h
    src/UIViewModule.cpp
)

# UIView is one aggregate QML module. Cross-cutting UI infrastructure belongs
# here; feature fragments append only their business-specific dependencies.
set(MODULE_DEPENDENCIES_PRIVATE
    AppContext
    LoggerUtils
    TranslatorManager
    UIEvents
    UIFabrication
    UIManager
    UIPlatformUtils
    UIQmlUtilities
    UIResourceAssetLoader
    UIResourceAssetRCCInterface
    UIResourceTranslation
    UIResourceLoader
    UIResourceLoaderManager
    UIViewCore
    UIViewModelSignalBridge
    UIViewSupport
    UIWindowUtilities
    UTComponent
    UTComposite
    ViewModelFactory
    Qt6::Core
    Qt6::Gui
    Qt6::Multimedia
    Qt6::Qml
    Qt6::Quick
    Qt6::Widgets
)

add_subdirectory(Shell)
add_subdirectory(Pages)
add_subdirectory(Features)

get_property(
    SUBMODULE_SOURCE_PRIVATE
    DIRECTORY "${UIVIEW_BUILD_ROOT}"
    PROPERTY UIVIEW_BUILD_SOURCES
)
get_property(
    MODULE_QML_TARGET_FILES
    DIRECTORY "${UIVIEW_BUILD_ROOT}"
    PROPERTY UIVIEW_BUILD_QML_FILES
)
get_property(
    MODULE_QML_TARGET_FILE_ALIAS_MAPPINGS
    DIRECTORY "${UIVIEW_BUILD_ROOT}"
    PROPERTY UIVIEW_BUILD_QML_ALIAS_MAPPINGS
)
get_property(
    MODULE_PRIVATE_INCLUDE_DIRS
    DIRECTORY "${UIVIEW_BUILD_ROOT}"
    PROPERTY UIVIEW_BUILD_PRIVATE_INCLUDE_DIRS
)
get_property(
    SUBMODULE_PRIVATE_DEPENDENCIES
    DIRECTORY "${UIVIEW_BUILD_ROOT}"
    PROPERTY UIVIEW_BUILD_PRIVATE_DEPENDENCIES
)

list(APPEND MODULE_SOURCE_PRIVATE ${SUBMODULE_SOURCE_PRIVATE})
list(APPEND MODULE_DEPENDENCIES_PRIVATE ${SUBMODULE_PRIVATE_DEPENDENCIES})

list(REMOVE_DUPLICATES MODULE_SOURCE_PRIVATE)
list(REMOVE_DUPLICATES MODULE_PRIVATE_INCLUDE_DIRS)
list(REMOVE_DUPLICATES MODULE_DEPENDENCIES_PRIVATE)

set(MODULE_DEPENDENCIES_PUBLIC)

set(MODULE_BUILD_INTERFACE_DIR
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)
set(MODULE_INSTALL_INTERFACE_DIR include)
set(MODULE_BUILD_PRIVATE_DIR src)

set(MODULE_QML_TARGET_URI
    ${MODULE_MODULE_NAME}
)

BuildQtModule(
    MODULE_NAME ${MODULE_MODULE_NAME}
    TARGET_SOURCE_PRIVATE ${MODULE_SOURCE_PRIVATE}
    TARGET_SOURCE_HEADER_BASE_DIR ${MODULE_HEADER_BASE_DIR}
    TARGET_SOURCE_PUBLIC_HEADER ${MODULE_HEADER_FILES}
    TARGET_ADD_LINK_LIBRARY_PRIVATE ${MODULE_DEPENDENCIES_PRIVATE}
    TARGET_ADD_LINK_LIBRARY_PUBLIC ${MODULE_DEPENDENCIES_PUBLIC}
    TARGET_INCLUDE_DIRECTORIES_BUILD_INTERFACE ${MODULE_BUILD_INTERFACE_DIR}
    TARGET_INCLUDE_DIRECTORIES_INSTALL_INTERFACE ${MODULE_INSTALL_INTERFACE_DIR}
    TARGET_INCLUDE_DIRECTORIES_PRIVATE ${MODULE_BUILD_PRIVATE_DIR} ${MODULE_PRIVATE_INCLUDE_DIRS}
    TARGET_PRIVATE_DEFINITIONS ${ADD_DEFINITIONS}
    IDE_FOLDER ${MODULE_IDE_FOLDER}
    QML_TARGET_URI ${MODULE_QML_TARGET_URI}
    QML_TARGET_FILES ${MODULE_QML_TARGET_FILES}
    QML_TARGET_FILE_ALIAS_MAPPINGS ${MODULE_QML_TARGET_FILE_ALIAS_MAPPINGS}
)

add_library(TemplateTool::QtDeploymentRoot ALIAS UIView)
