include(BuildModule)

set(MODULE_MODULE_NAME PermissionPlatform)
set(MODULE_IDE_FOLDER framework/utilities/platform)
set(MODULE_HEADER_BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/include)

set(MODULE_HEADER_FILES
    include/ucf/utilities/PermissionPlatform/PermissionPlatformExport.h
    include/ucf/utilities/PermissionPlatform/PermissionPlatform.h
)

set(MODULE_DEPENDENCIES_PRIVATE)
if(APPLE)
    set(MODULE_SOURCE_PRIVATE src/macos/PermissionPlatform_Mac.mm)
    find_library(PERMISSION_PLATFORM_COREGRAPHICS CoreGraphics REQUIRED)
    find_library(PERMISSION_PLATFORM_AVFOUNDATION AVFoundation REQUIRED)
    find_library(PERMISSION_PLATFORM_APPKIT AppKit REQUIRED)
    find_library(PERMISSION_PLATFORM_FOUNDATION Foundation REQUIRED)
    list(APPEND MODULE_DEPENDENCIES_PRIVATE
        ${PERMISSION_PLATFORM_COREGRAPHICS}
        ${PERMISSION_PLATFORM_AVFOUNDATION}
        ${PERMISSION_PLATFORM_APPKIT}
        ${PERMISSION_PLATFORM_FOUNDATION}
        objc
    )
else()
    set(MODULE_SOURCE_PRIVATE src/PermissionPlatform_Unsupported.cpp)
endif()

set(MODULE_PUBLIC_DEFINITIONS)
set(MODULE_PRIVATE_DEFINITIONS)
if(UTILITY_LIB_TYPE STREQUAL "STATIC_LIB")
    set(MODULE_PUBLIC_DEFINITIONS PERMISSION_PLATFORM_STATIC)
else()
    set(MODULE_PRIVATE_DEFINITIONS PERMISSION_PLATFORM_SHARED)
endif()

BuildModule(
    ${UTILITY_LIB_TYPE}
    MODULE_NAME ${MODULE_MODULE_NAME}
    TARGET_SOURCE_PRIVATE ${MODULE_SOURCE_PRIVATE}
    TARGET_SOURCE_HEADER_BASE_DIR ${MODULE_HEADER_BASE_DIR}
    TARGET_SOURCE_PUBLIC_HEADER ${MODULE_HEADER_FILES}
    TARGET_ADD_LINK_LIBRARY_PRIVATE ${MODULE_DEPENDENCIES_PRIVATE}
    TARGET_INCLUDE_DIRECTORIES_BUILD_INTERFACE ${MODULE_HEADER_BASE_DIR}
    TARGET_INCLUDE_DIRECTORIES_INSTALL_INTERFACE include
    TARGET_INCLUDE_DIRECTORIES_PRIVATE src
    TARGET_PUBLIC_DEFINITIONS ${MODULE_PUBLIC_DEFINITIONS}
    TARGET_PRIVATE_DEFINITIONS ${MODULE_PRIVATE_DEFINITIONS}
    IDE_FOLDER ${MODULE_IDE_FOLDER}
)
